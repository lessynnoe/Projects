<script>

let players = [];
let parking = 0;
let history = [];
let logs = [];
let startPlayer = null;

const colors=[
"#1565c0",
"#c62828",
"#2e7d32",
"#ef6c00",
"#7b1fa2",
"#00838f",
"#ad1457",
"#5d4037"
];



function addPlayer(){

const name=document.getElementById("playerName").value.trim();
const money=Number(document.getElementById("startMoney").value);

if(!name){
alert("Spielername eingeben.");
return;
}

if(players.some(p=>p.name.toLowerCase()===name.toLowerCase())){
alert("Spieler existiert bereits.");
return;
}

saveHistory();

players.push({
id:Date.now()+Math.random(),
name,
money,
color:colors[players.length%colors.length]
});

document.getElementById("playerName").value="";

addLog(`👤 ${name} startet mit ${money} M.`,"neutral");

saveGame();
render();

}



function render(){

const container=document.getElementById("players");

container.innerHTML="";

players.forEach(player=>{

const div=document.createElement("div");

div.className="player";
div.style.borderTopColor=player.color;

let badge="";

if(player.id===startPlayer){
badge=`<div class="startBadge">STARTSPIELER</div>`;
}

div.innerHTML=`

${badge}

<h3>${escapeHTML(player.name)}</h3>

<div class="money">
${formatMoney(player.money)} M
</div>

<input
type="number"
id="amount-${player.id}"
placeholder="Betrag">

<div class="quick">

<button onclick="setPlayerAmount(${player.id},10)">10</button>
<button onclick="setPlayerAmount(${player.id},50)">50</button>
<button onclick="setPlayerAmount(${player.id},100)">100</button>
<button onclick="setPlayerAmount(${player.id},200)">200</button>
<button onclick="setPlayerAmount(${player.id},500)">500</button>

</div>

<div class="player-buttons">

<button class="green" onclick="changeMoney(${player.id},1)">
🏦 Bank +
</button>

<button class="red" onclick="changeMoney(${player.id},-1)">
🏦 Bank -
</button>

<button class="green" onclick="passGo(${player.id})">
🏁 LOS +200
</button>

<button class="orange" onclick="sendToParking(${player.id})">
🚗 Freiparken
</button>

<button class="purple" onclick="bail(${player.id})">
🚔 Kaution -50
</button>

<button class="dark" onclick="removePlayer(${player.id})">
✖ Entfernen
</button>

</div>

`;

container.appendChild(div);

});

document.getElementById("parkingMoney").innerText=formatMoney(parking);
document.getElementById("playerCount").innerText=players.length;

const total=players.reduce((sum,p)=>sum+p.money,0);

document.getElementById("totalMoney").innerText=
formatMoney(total)+" M";

renderLog();

updateSelects();

}



function setPlayerAmount(id,value){

document.getElementById(`amount-${id}`).value=value;

}



function changeMoney(id,direction){

const player=players.find(p=>p.id===id);

const input=document.getElementById(`amount-${id}`);

const amount=Number(input.value);

if(!amount||amount<=0){
alert("Betrag eingeben.");
return;
}

saveHistory();

player.money+=amount*direction;

if(direction===1){

addLog(
`🏦 Bank → ${player.name}: +${amount} M`,
"positive"
);

}else{

addLog(
`${player.name} → 🏦 Bank: ${amount} M`,
"negative"
);

}

input.value="";

saveGame();
render();

}



function passGo(id){

const player=players.find(p=>p.id===id);

saveHistory();

player.money+=200;

addLog(
`🏁 ${player.name} geht über LOS und erhält 200 M.`,
"positive"
);

saveGame();
render();

}



function quickAction(amount,type){

const id=Number(document.getElementById("quickPlayer").value);

if(!id){
alert("Spieler auswählen.");
return;
}

const player=players.find(p=>p.id===id);

saveHistory();

player.money+=amount;

if(amount>0){

addLog(
`${type==="LOS"?"🏁":"🎁"} ${player.name}: +${amount} M (${type})`,
"positive"
);

}else{

addLog(
`🏛 ${player.name}: ${amount} M (${type})`,
"negative"
);

}

saveGame();
render();

}



function transferMoney(){

const from=Number(document.getElementById("transferFrom").value);
const to=Number(document.getElementById("transferTo").value);
const amount=Number(document.getElementById("transferAmount").value);

if(!from||!to||!amount||amount<=0){

alert("Spieler und Betrag auswählen.");

return;

}

if(from===to){

alert("Gleicher Spieler.");

return;

}

const sender=players.find(p=>p.id===from);
const receiver=players.find(p=>p.id===to);

saveHistory();

sender.money-=amount;
receiver.money+=amount;

addLog(
`💸 ${sender.name} → ${receiver.name}: ${amount} M`,
"neutral"
);

document.getElementById("transferAmount").value="";

saveGame();
render();

}



function setTransfer(value){

document.getElementById("transferAmount").value=value;

}



function payRent(){

const from=Number(document.getElementById("rentFrom").value);
const to=Number(document.getElementById("rentTo").value);
const amount=Number(document.getElementById("rentAmount").value);

if(!from||!to||!amount||amount<=0){

alert("Mieter, Besitzer und Miete auswählen.");

return;

}

if(from===to){

alert("Mieter und Besitzer müssen verschieden sein.");

return;

}

const renter=players.find(p=>p.id===from);
const owner=players.find(p=>p.id===to);

saveHistory();

renter.money-=amount;
owner.money+=amount;

addLog(
`🏠 ${renter.name} zahlt ${owner.name} ${amount} M Miete.`,
"neutral"
);

document.getElementById("rentAmount").value="";

saveGame();
render();

}



function setRent(value){

document.getElementById("rentAmount").value=value;

}

const player=players.find(p=>p.id===id);

const amount=Number(input.value);

if(!amount||amount<=0){

alert("Betrag eingeben.");

}

const id=Number(document.getElementById("quickPlayer").value);

if(!id){

alert("Spieler auswählen.");

return;

}

const player=players.find(p=>p.id===id);

saveHistory();

);

function collectParking(){

const id=Number(document.getElementById("parkingWinner").value);

if(!id){

alert("Spieler auswählen.");

return;

}

if(parking<=0){

alert("Freiparken ist leer.");

return;

}

const player=players.find(p=>p.id===id);

saveHistory();

const win=parking;

player.money+=parking;
parking=0;

addLog(
`🎉 ${player.name} kassiert ${win} M aus Freiparken!`,
"positive"
);

saveGame();
render();

}



function bail(id){

const player=players.find(p=>p.id===id);

saveHistory();

player.money-=50;

`🚔 ${player.name} bezahlt 50 M Kaution.`,
"negative"
);

}



function birthday(){

const id=Number(document.getElementById("quickPlayer").value);

if(!id){

alert("Geburtstagskind auswählen.");

return;

}

const birthdayPlayer=players.find(p=>p.id===id);

if(players.length<2){

alert("Mindestens 2 Spieler erforderlich.");

return;

}

saveHistory();

players.forEach(player=>{

if(player.id!==id){

player.money-=10;
birthdayPlayer.money+=10;

}

});

addLog(
`🎂 ${birthdayPlayer.name} hat Geburtstag. Jeder zahlt 10 M.`,
"positive"
);

saveGame();
render();

}



function bankPaysEveryone(amount){

if(players.length===0)return;

saveHistory();

players.forEach(player=>{
player.money+=amount;
});

addLog(
`🏦 Die Bank zahlt jedem Spieler ${amount} M.`,
"positive"
);

saveGame();
render();

}



function everyonePaysBank(amount){

if(players.length===0)return;

saveHistory();

players.forEach(player=>{
player.money-=amount;
});

addLog(
`🏦 Jeder Spieler zahlt ${amount} M an die Bank.`,
"negative"
);

saveGame();
render();

}



function randomStartPlayer(){

if(players.length===0){

alert("Zuerst Spieler hinzufügen.");

return;

}

const random=players[Math.floor(Math.random()*players.length)];

startPlayer=random.id;

addLog(
`🎲 ${random.name} wurde als Startspieler ausgelost.`,
"neutral"
);

saveGame();
render();

}



function removePlayer(id){

const player=players.find(p=>p.id===id);

if(!confirm(`${player.name} wirklich entfernen?`)){
return;
}

saveHistory();

players=players.filter(p=>p.id!==id);

if(startPlayer===id){
startPlayer=null;
}

addLog(
`❌ ${player.name} wurde entfernt.`,
"negative"
);

saveGame();
render();

}



function updateSelects(){

const configs=[
["quickPlayer","Spieler auswählen..."],
["transferFrom","Von Spieler..."],
["transferTo","An Spieler..."],
["rentFrom","Mieter..."],
["rentTo","Besitzer..."],
["parkingWinner","Spieler auswählen..."]
];

configs.forEach(config=>{

const select=document.getElementById(config[0]);

const old=select.value;

select.innerHTML=
`<option value="">${config[1]}</option>`;

players.forEach(player=>{

const option=document.createElement("option");

option.value=player.id;

option.textContent=
`${player.name} (${formatMoney(player.money)} M)`;

select.appendChild(option);

});

select.value=old;

});

}



function addLog(text,type="neutral"){

const time=new Date().toLocaleTimeString(
"de-DE",
{
hour:"2-digit",
minute:"2-digit"
}
);

logs.unshift({
text,
type,
time
});

if(logs.length>100){
logs=logs.slice(0,100);
}

}



function renderLog(){

const log=document.getElementById("log");

if(logs.length===0){

log.innerHTML=
`<div class="logEntry logNeutral">
Noch keine Transaktionen.
</div>`;

return;

}

log.innerHTML=logs.map(entry=>{

let cls="logNeutral";

if(entry.type==="positive"){
cls="logPositive";
}

if(entry.type==="negative"){
cls="logNegative";
}

return`
<div class="logEntry ${cls}">
<strong>${entry.time}</strong> –
${escapeHTML(entry.text)}
</div>
`;

}).join("");

}



function saveHistory(){

history.push({

players:JSON.parse(JSON.stringify(players)),
parking,
logs:JSON.parse(JSON.stringify(logs)),
startPlayer

});

if(history.length>20){
history.shift();
}

}



function undo(){

if(history.length===0){

alert("Nichts zum Rückgängig machen.");

return;

}

const previous=history.pop();

players=previous.players;
parking=previous.parking;
logs=previous.logs;
startPlayer=previous.startPlayer;

addLog(
"↩ Letzte Aktion wurde rückgängig gemacht.",
"neutral"
);

saveGame();
render();

}



function saveGame(){

localStorage.setItem(
"monopolyDigitalBank",
JSON.stringify({
players,
parking,
logs,
startPlayer
})
);

}



function loadGame(){

const saved=
localStorage.getItem("monopolyDigitalBank");

if(saved){

try{

const data=JSON.parse(saved);

players=data.players||[];
parking=data.parking||0;
logs=data.logs||[];
startPlayer=data.startPlayer||null;

}catch(e){

console.log("Spielstand konnte nicht geladen werden.");

}

}

render();

}



function resetGame(){

if(!confirm(
"Wirklich ein komplett neues Monopoly-Spiel starten?"
)){
return;
}

players=[];
parking=0;
history=[];
logs=[];
startPlayer=null;

localStorage.removeItem("monopolyDigitalBank");

addLog(
"🎲 Neues Monopoly-Spiel gestartet.",
"neutral"
);

render();

}



function formatMoney(number){

return Number(number).toLocaleString("de-DE");

}



function escapeHTML(text){

return String(text)
.replaceAll("&","&amp;")
.replaceAll("<","&lt;")
.replaceAll(">","&gt;")
.replaceAll('"',"&quot;")
.replaceAll("'","&#039;");

}



loadGame();
